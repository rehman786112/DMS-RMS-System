############### Customers Database ################


CREATE TABLE IF NOT EXISTS customers (
    cus_code INT PRIMARY KEY AUTO_INCREMENT,
    cus_name VARCHAR(100) NOT NULL,
    cus_email VARCHAR(100) NOT NULL UNIQUE,
    cus_address VARCHAR(200) NOT NULL,
    cus_city VARCHAR(100) NOT NULL,
    cus_country VARCHAR(100) DEFAULT 'Pakistan',
    cus_type VARCHAR(50),
    cus_cnic VARCHAR(50),
    cus_area VARCHAR(100),
    cus_sub_area VARCHAR(100),
    cus_credit_limit DECIMAL(10,2) DEFAULT 0,
    cus_whatsapp VARCHAR(20),
    phone varchar(100),
    cus_date DATE DEFAULT (CURRENT_DATE)
);

cus_code	int	NO	PRI		auto_increment
cus_name	varchar(100)	NO			
cus_email	varchar(100)	NO	UNI		
cus_address	varchar(200)	NO			
cus_city	varchar(100)	NO	MUL		
cus_country	varchar(100)	YES		Pakistan	
cus_type	varchar(50)	YES	MUL		
cus_cnic	varchar(50)	YES			
cus_area	varchar(100)	YES	MUL		
cus_sub_area	varchar(100)	YES	MUL		
cus_credit_limit	decimal(10,2)	YES	MUL	0.00	
cus_date	date	YES		curdate()	DEFAULT_GENERATED
phone	varchar(100)	YES			
cus_whatsapp	varchar(20)	YES

##################    AREA & SUB-AREA DATABASE  ##################

CREATE TABLE IF NOT EXISTS areas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    area_name VARCHAR(100) NOT NULL UNIQUE
);
CREATE TABLE IF NOT EXISTS sub_areas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sub_area_name VARCHAR(100) NOT NULL,
    area_id INT,
    area_name VARCHAR(100),
    FOREIGN KEY (area_id) REFERENCES areas(id) ON DELETE CASCADE,
    FOREIGN KEY (area_name) REFERENCES areas(area_name) ON DELETE CASCADE
);
